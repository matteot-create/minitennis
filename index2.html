// =============================== // FIREBASE INIT // =============================== import { initializeApp } from "https://www.gstatic.com/firebasejs/10.7.1/firebase-app.js"; import { getFirestore, doc, setDoc, deleteDoc, collection, addDoc, onSnapshot, query, orderBy } from "https://www.gstatic.com/firebasejs/10.7.1/firebase-firestore.js"; const firebaseConfig = { apiKey: "YOUR_KEY", authDomain: "YOUR_DOMAIN", projectId: "YOUR_PROJECT", storageBucket: "YOUR_BUCKET", messagingSenderId: "YOUR_SENDER", appId: "YOUR_APP" }; const app = initializeApp(firebaseConfig); const db = getFirestore(app); // =============================== // GLOBAL STATE // =============================== let currentUser = prompt("Enter your name:"); let currentRoom = "general"; // "general" or private roomId // =============================== // DOM ELEMENTS // =============================== const messagesDiv = document.getElementById("messages"); const msgInput = document.getElementById("msgInput"); const sendBtn = document.getElementById("sendBtn"); const membersList = document.getElementById("membersList"); const chatTitle = document.getElementById("chatTitle"); const leavePrivateBtn = document.getElementById("leavePrivateBtn"); const inviteBanner = document.getElementById("inviteBanner"); const inviteText = document.getElementById("inviteText"); const joinPrivateBtn = document.getElementById("joinPrivateBtn"); // =============================== // PRESENCE SYSTEM // =============================== async function setPresence() { await setDoc(doc(db, "presence", currentUser), { name: currentUser, timestamp: Date.now() }); } setPresence(); setInterval(setPresence, 20000); // =============================== // LOAD MEMBERS LIST // =============================== onSnapshot(collection(db, "presence"), (snapshot) => { membersList.innerHTML = ""; snapshot.forEach((docSnap) => { const user = docSnap.id; // label "in private chat" se l'utente è nella stanza corrente let privateLabel = ""; if (currentRoom !== "general" && currentRoom.includes(user)) { privateLabel = `(in private chat)`; } const div = document.createElement("div"); div.className = "member"; div.innerHTML = ` ${user} ${privateLabel} Available `; membersList.appendChild(div); }); // click su Available → invito alla chat privata document.querySelectorAll(".availableBtn").forEach(btn => { btn.onclick = () => { const targetUser = btn.dataset.user; // niente self‑invite if (!targetUser || targetUser === currentUser) { console.warn("Invalid private chat target:", targetUser); return; } createPrivateInvite(targetUser); }; }); }); // =============================== // PRIVATE CHAT INVITE SYSTEM // =============================== function createPrivateInvite(targetUser) { // Prevent self-invite if (!targetUser || targetUser === currentUser) { console.warn("Invalid private chat target:", targetUser); return; } const roomId = [currentUser, targetUser].sort().join("_"); // invito salvato per l'utente B setDoc(doc(db, "privateInvites", targetUser), { from: currentUser, roomId: roomId, timestamp: Date.now() }); // A entra subito nella stanza privata joinPrivateChat(roomId); } // listener inviti per chi è B onSnapshot(doc(db, "privateInvites", currentUser), (docSnap) => { if (docSnap.exists()) { const data = docSnap.data(); inviteBanner.style.display = "block"; inviteText.innerText = `${data.from} wants to chat privately with you`; joinPrivateBtn.onclick = () => { joinPrivateChat(data.roomId); deleteDoc(doc(db, "privateInvites", currentUser)); inviteBanner.style.display = "none"; }; } else { inviteBanner.style.display = "none"; } }); // =============================== // SWITCH TO PRIVATE CHAT // =============================== function joinPrivateChat(roomId) { currentRoom = roomId; chatTitle.innerText = `Private Chat (${roomId})`; leavePrivateBtn.style.display = "inline-block"; loadMessages(); } // =============================== // LEAVE PRIVATE CHAT // =============================== leavePrivateBtn.onclick = () => { currentRoom = "general"; chatTitle.innerText = "General Chat"; leavePrivateBtn.style.display = "none"; loadMessages(); }; // =============================== // LOAD MESSAGES (GENERAL OR PRIVATE) // =============================== let unsubscribeMessages = null; function loadMessages() { if (unsubscribeMessages) unsubscribeMessages(); messagesDiv.innerHTML = ""; const col = currentRoom === "general" ? collection(db, "chat") : collection(db, "privateChats", currentRoom, "messages"); const q = query(col, orderBy("timestamp")); unsubscribeMessages = onSnapshot(q, (snapshot) => { messagesDiv.innerHTML = ""; snapshot.forEach((docSnap) => { const msg = docSnap.data(); const div = document.createElement("div"); div.innerHTML = `${msg.user}: ${msg.text}`; messagesDiv.appendChild(div); }); messagesDiv.scrollTop = messagesDiv.scrollHeight; }); } loadMessages(); // =============================== // SEND MESSAGE // =============================== sendBtn.onclick = async () => { const text = msgInput.value.trim(); if (!text) return; const col = currentRoom === "general" ? collection(db, "chat") : collection(db, "privateChats", currentRoom, "messages"); await addDoc(col, { user: currentUser, text: text, timestamp: Date.now() }); msgInput.value = ""; };
